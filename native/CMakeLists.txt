cmake_minimum_required(VERSION 3.20)
project(BullseyeGeometry LANGUAGES CXX)
if((NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    message(FATAL_ERROR "The audited engine ABIs are Windows x64 and Linux x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(BullseyeGeometry SHARED hitbox_snapshot.cpp)
enable_testing()
add_executable(HitboxSnapshotTests tests/hitbox_snapshot_tests.cpp hitbox_snapshot.cpp)
target_include_directories(HitboxSnapshotTests PRIVATE .)
foreach(target BullseyeGeometry HitboxSnapshotTests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic)
    endif()
endforeach()
add_test(NAME HitboxSnapshotTests COMMAND HitboxSnapshotTests)
